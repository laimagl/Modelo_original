from plotly.subplots import make_subplots

df_plot = df_5m['2020-03-01':'2020-04-01']
df_plot = df_plot.copy()
df_plot['Rango'] = df_plot['BH'] - df_plot['BL']
df_plot['Spread'] = df_plot['AO'] - df_plot['BO']

fig = make_subplots(rows=3, cols=1, 
                    shared_xaxes=True,
                    subplot_titles=(
                        'Precio de Cierre | Bid',
                        'Rango de Vela | Volatilidad (BH - BL)',
                        'Spread | Liquidez (AO - BO)'
                    ))

# Precio de cierre
fig.add_trace(go.Scatter(
    x=df_plot.index, y=df_plot['BC'],
    mode='lines', name='Cierre Bid',
    line=dict(color='royalblue', width=1)
), row=1, col=1)

# Rango
fig.add_trace(go.Scatter(
    x=df_plot.index, y=df_plot['Rango'],
